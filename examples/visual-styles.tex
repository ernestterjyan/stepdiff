\section*{Display modes}

\stylecaption{Notes display}
\[
\begin{stepdiff}[display=notes, theme=soft, layout=lecture, reason-style=muted]
\step{S_n = 1 + 2 + \cdots + n}
\step[reason={pair terms}]{2S_n = n(n + 1)}
\step[reason={final formula}, diff=all, tag=final]{S_n = \frac{n(n + 1)}{2}}
\end{stepdiff}
\]

\stylecaption{Slide display}
\[
\begin{stepdiff}[display=slide, theme=soft, layout=lecture, frame=true]
\step{f(x) = x^2 + 2x}
\step[reason={differentiate}]{f'(x) = 2x + 2}
\step[reason={evaluate}, tag=final]{f'(3) = 8}
\end{stepdiff}
\]

\stylecaption{Focus display}
\[
\begin{stepdiff}[display=focus, theme=focus, reason-style=badge]
\step{(x + 1)^2}
\step[reason={expand}]{x^2 + 2x + 1}
\step[reason={final}, tag=final]{x^2 + 2x + 1}
\end{stepdiff}
\]

\section*{Reason labels and frames}

\stylecaption{Muted reasons}
\[
\begin{stepdiff}[display=notes, theme=soft, reason-style=muted]
\step{u = v}
\step[reason={substitute known value}]{u = w}
\step[reason={conclude}, tag=final]{u = w}
\end{stepdiff}
\]

\stylecaption{Badge reasons}
\[
\begin{stepdiff}[display=focus, theme=focus, reason-style=badge]
\step{g(x) = x^2\sin x}
\step[reason={product rule}]{g'(x) = 2x\sin x + x^2\cos x}
\step[reason={evaluate}, tag=final]{g'(0) = 0}
\end{stepdiff}
\]

\stylecaption{Framed block}
\[
\begin{stepdiff}[display=notes, theme=soft, frame=true]
\step{x^2 - 1 = (x - 1)(x + 1)}
\step[reason={expand}]{x^2 - 1 = x^2 - 1}
\step[reason={same identity}, tag=final]{x^2 - 1 = (x - 1)(x + 1)}
\end{stepdiff}
\]

\section*{Customization hook}

\stylecaption{Custom changed style}
\begingroup
\renewcommand{\SDchanged}[1]{\begingroup\color{red!65!black}#1\endgroup}
\renewcommand{\SDreason}[1]{\normalfont\scriptsize\sffamily\color{black!55}#1}
\[
\begin{stepdiff}[display=notes, theme=soft, frame=true]
\step{p(x) = x^2 + 2x + 1}
\step[reason={custom hook}]{p(x) = (x + 1)^2}
\step[reason={perfect square}, diff=all, tag=final]{p(x) = (x + 1)^2}
\end{stepdiff}
\]
\endgroup
