\documentclass{article}
\usepackage{xcolor}
\usepackage{stepdiff}
\begin{document}

\[
\begin{stepdiff}[display=notes, theme=soft, reason-style=muted]
\step{x+1 = 2}
\step[reason={subtract 1}]{x = 1}
\end{stepdiff}
\]

\[
\begin{stepdiff}[display=focus, theme=focus, highlight=background]
\step{a_n \le b_n}
\step[reason={take limits}]{\lim a_n \le \lim b_n}
\step[reason={conclude}, tag=final]{L \le M}
\end{stepdiff}
\]

\[
\begin{stepdiff}[display=slide, layout=wide, highlight=color]
\step{P \Rightarrow Q}
\step[reason={strengthen hypothesis}]{P \wedge R \Rightarrow Q}
\step[reason={iterate}, diff=all, tag=final]{P \wedge R \Longrightarrow Q}
\end{stepdiff}
\]

\[
\begin{stepdiff}[frame=true, reason-style=badge]
\step{x < y = z}
\step[reason={estimate}]{x \approx y}
\step[reason={identify}, tag=final]{x \equiv y}
\end{stepdiff}
\]

\end{document}
