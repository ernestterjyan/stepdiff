\section*{A compact worked expansion}

\[
\begin{stepdiff}[color-mode=typed, frame=true]
\step{(x + 2)(x + 3)}
\step[reason={distribute the first factor}]{x(x + 3) + 2(x + 3)}
\step[reason={expand products}]{x^2 + 3x + 2x + 6}
\step[reason={collect like terms}]{x^2 + 5x + 6}
\step[reason={standard form}, diff=all, tag=final]{x^2 + 5x + 6}
\end{stepdiff}
\]

\section*{Relation-aware alignment}

\[
\begin{stepdiff}[display=focus, theme=focus, color-mode=teaching, reason-style=badge, frame=true, legend=true]
\step{a_n \le b_n}
\step[reason={take limits}]{\lim a_n \le \lim b_n}
\step[reason={identify the limits}, diff=all, tag=final]{L \le M}
\end{stepdiff}
\]

\section*{Focus mode for a final value}

\[
\begin{stepdiff}[display=focus, theme=focus, reason-style=badge, frame=true]
\step{f(x) = x^3 + 2x}
\step[reason={differentiate}]{f'(x) = 3x^2 + 2}
\step[reason={evaluate at x = 2}]{f'(2) = 3 \cdot 2^2 + 2}
\step[reason={simplify}, diff=all, tag=final]{f'(2) = 14}
\end{stepdiff}
\]

\section*{A polished framed derivation}

\[
\begin{stepdiff}[display=notes, theme=soft, color-mode=typed, frame=true]
\step{x^2 - 1 = x^2 - 1}
\step[reason={write one as a square}]{x^2 - 1 = x^2 - 1^2}
\step[reason={factor difference of squares}, diff=all, tag=final]{x^2 - 1 = (x - 1)(x + 1)}
\end{stepdiff}
\]
