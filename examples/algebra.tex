\section*{Expansion}

\[
\begin{stepdiff}
\step{a(x + b)}
\step[reason={distribute a}]{ax + ab}
\step[reason={commute terms}]{ab + ax}
\end{stepdiff}
\]

\section*{Factorization}

\[
\begin{stepdiff}
\step{x^2 + 2x + 1}
\step[reason={recognize a square}]{x^2 + 2x + 1 = (x + 1)^2}
\step[reason={substitute x = 3}]{3^2 + 2 \cdot 3 + 1 = (3 + 1)^2}
\step[reason={evaluate}]{16 = 16}
\end{stepdiff}
\]
