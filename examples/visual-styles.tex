\section*{Display modes}

\[
\begin{stepdiff}[display=notes, theme=soft, reason-style=muted]
\step{S_n = 1 + 2 + \cdots + n}
\step[reason={pair terms}]{2S_n = n(n+1)}
\step[reason={final formula}, diff=all, tag=final]{S_n = \frac{n(n+1)}{2}}
\end{stepdiff}
\]

\[
\begin{stepdiff}[display=slide, theme=soft, frame=true]
\step{f(x)=x^2+2x}
\step[reason={differentiate}]{f'(x)=2x+2}
\step[reason={evaluate}, tag=final]{f'(3)=8}
\end{stepdiff}
\]

\[
\begin{stepdiff}[display=focus, theme=focus, reason-style=badge]
\step{(x+1)^2}
\step[reason={expand}]{x^2+2x+1}
\step[reason={final}, tag=final]{x^2+2x+1}
\end{stepdiff}
\]

\section*{Themes}

\[
\begin{stepdiff}[theme=soft]
\step{a=b}
\step[reason={soft}]{a=c}
\end{stepdiff}
\qquad
\begin{stepdiff}[theme=minimal]
\step{a=b}
\step[reason={minimal}]{a=c}
\end{stepdiff}
\]

\[
\begin{stepdiff}[theme=focus, display=focus]
\step{a=b}
\step[reason={focus}, tag=final]{a=c}
\end{stepdiff}
\]

\section*{Highlight modes}

\[
\begin{stepdiff}[highlight=background]

\step[reason={background}]{x=2}
\end{stepdiff}
\qquad
\begin{stepdiff}[highlight=underline]
\step{x=1}
\step[reason={underline}]{x=2}
\end{stepdiff}
\]

\[
\begin{stepdiff}[highlight=color]
\step{x=1}
\step[reason={color}]{x=2}
\end{stepdiff}
\qquad
\begin{stepdiff}[highlight=none]
\step{x=1}
\step[reason={none}]{x=2}
\end{stepdiff}
\]

\section*{Reason styles and framing}

\[
\begin{stepdiff}[reason-style=plain]
\step{u=v}
\step[reason={plain}]{u=w}
\end{stepdiff}
\qquad
\begin{stepdiff}[reason-style=muted]
\step{u=v}
\step[reason={muted}]{u=w}
\end{stepdiff}
\]

\[
\begin{stepdiff}[reason-style=badge, frame=true]
\step{u=v}
\step[reason={badge}]{u=w}
\step[reason={final badge}, tag=final]{u=z}
\end{stepdiff}
\]
