\documentclass{article}
\usepackage{xcolor}
\usepackage{stepdiff}
\begin{document}

\[
\begin{stepdiff}[display=notes, theme=soft, highlight=background, reason-style=muted, frame=true]
\step{S_n = 1 + 2 + \cdots + n}
\step[reason={pair terms}]{2S_n = n(n+1)}
\step[reason={final formula}, diff=all, tag=final]{S_n = \frac{n(n+1)}{2}}
\end{stepdiff}
\]

\[
\begin{stepdiff}[display=slide, theme=minimal, highlight=underline, reason-style=plain]
\step{x^2 - 1 = (x-1)(x+1)}
\step[reason={expand}, diff=false]{x^2 - 1 = x^2 - 1}
\step[reason={restore}, tag=final]{x^2 - 1 = (x-1)(x+1)}
\end{stepdiff}
\]

\[
\begin{stepdiff}[display=focus, theme=focus, highlight=color, reason-style=badge]
\step{f(x)=x^2+2x}
\step[reason={differentiate}]{f'(x)=2x+2}
\step[reason={evaluate}, tag=final]{f'(3)=8}
\end{stepdiff}
\]

\[
\begin{stepdiff}[highlight=none, layout=wide, reason-style=muted]
\step{P \Rightarrow Q}
\step[reason={add hypothesis}]{P \wedge R \Rightarrow Q}
\end{stepdiff}
\]

\[
\begin{stepdiff}[layout=compact, frame=false]
\step{a=b}
\step[reason={compact}]{a=c}
\end{stepdiff}
\]
