\section*{Diff modes}

\[
\begin{stepdiff}
\step{x^2 - 1 = (x - 1)(x + 1)}
\step[reason={automatic diff}, diff=auto]{x^2 - 1 = x^2 - 1}
\step[reason={suppress noisy highlight}, diff=false]{x^2 - 1 = x^2 - 1}
\step[reason={equivalent spelling}, diff=none]{x^2 - 1 = x^2 - 1}
\step[reason={important identity}, diff=all]{x^2 - 1 = (x - 1)(x + 1)}
\end{stepdiff}
\]

\section*{Custom highlighting}

\begingroup
\renewcommand{\SDchanged}[1]{\begingroup\color{red!70!black}#1\endgroup}
\[
\begin{stepdiff}
\step{p(x) = x^2 + 2x + 1}
\step[reason={custom highlight style}]{p(x) = (x + 1)^2}
\end{stepdiff}
\]
\endgroup

\section*{Hide reasons}

\SDhidereasons
\[
\begin{stepdiff}
\step{g(x) = x^2}
\step[reason={hidden in this display}]{g'(x) = 2x}
\end{stepdiff}
\]
