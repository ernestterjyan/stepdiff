\step<1->{f(x) = x^2 + 3x}
\step<2->[reason={differentiate}]{f'(x) = 2x + 3}
\step<3->[reason={evaluate at x = 3}]{f'(3) = 2 \cdot 3 + 3}
\step<4->[reason={final value}, diff=all, tag=final]{f'(3) = 9}
\end{stepdiff}
\]
